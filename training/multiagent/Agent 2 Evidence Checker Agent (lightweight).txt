🔹 Updated EvidenceCheckerAgent (Hybrid)
FUNCTION EvidenceCheckerAgent(text, language, prediction) -> EvidenceCheck:

    key_terms = symptom_dictionary[prediction.label][language]
    matched_terms = []

    FOR term IN key_terms:
        IF term EXISTS IN text:
            matched_terms.append(term)

    hit_count = LENGTH(matched_terms)

    // ---- Rule-based decision ----
    IF hit_count >= 3:
        support = SUPPORTED
        rationale = "Multiple key symptoms found: " + join(matched_terms)
        missing_questions = []
        RETURN EvidenceCheck(support, rationale, missing_questions)

    ELSE IF hit_count == 0:
        support = CONTRADICTED
        rationale = "No expected symptoms for " + prediction.label + " found in text."
        missing_questions = propose_questions(prediction.label)
        RETURN EvidenceCheck(support, rationale, missing_questions)

    // ---- Uncertain case → call LLM ----
    ELSE:
        llm_result = LLM_EvidenceCheck(text, prediction.label)

        support = llm_result.support_level
        rationale = llm_result.rationale
        missing_questions = llm_result.missing_questions

        RETURN EvidenceCheck(support, rationale, missing_questions)

🔹 LLM Evidence Check (Single Call Only)
FUNCTION LLM_EvidenceCheck(text, predicted_label):

    PROMPT = """
    You are a clinical evidence verification assistant.

    Task:
    - Check whether the following clinical note supports the diagnosis: {predicted_label}
    - Base your judgment ONLY on the given text.
    - Do NOT invent symptoms.

    Clinical Note:
    {text}

    Output format (strict):
    Support Level: SUPPORTED / WEAK / CONTRADICTED
    Rationale: short explanation quoting symptoms if present
    Missing Questions: list up to 3 clarification questions
    """

    RESPONSE = call_LLM(PROMPT)

    PARSE RESPONSE INTO:
        support_level
        rationale
        missing_questions

    RETURN {support_level, rationale, missing_questions}
